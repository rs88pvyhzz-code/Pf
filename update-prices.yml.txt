name: Update prices
on:
  schedule:
    - cron: "45 16 * * 1-5"   # after the London close
    - cron: "30 21 * * 1-5"   # after the New York close
  workflow_dispatch: {}       # lets you run it by hand from the Actions tab
permissions:
  contents: write
jobs:
  prices:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: python fetch_prices.py
      - name: Commit prices
        run: |
          git config user.name "price-bot"
          git config user.email "price-bot@users.noreply.github.com"
          git add prices.json
          git diff --cached --quiet || git commit -m "Update prices"
          git push
